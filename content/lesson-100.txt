文档分块函数
def split_document(file_path):
    """读取文档并按空行分块"""
    with open(file_path, 'r', encoding='utf-8') as f:
        content = f.read()
    return [chunk.strip() for chunk in content.split('\n\n') if chunk.strip()]
按空行切分：split('\n\n') 利用数据文件的结构化特点
按空行切分
过滤空块：if chunk.strip() 避免空内容进入数据库
过滤空块
简单高效：对于结构化数据，简单的字符串切分就够了。如果是普通文档（如PDF、Word），可能
需要使用NLP工具（spaCy、LangChain等）进行智能分句。但我们给出的示例数据已经用【】标记
和空行预处理好了

all_chunks, all_metadata, all_ids = [], [], []
txt_files = [f for f in os.listdir(DATA_DIR) if f.endswith('.txt')]
for filename in txt_files:
    character_name = filename.replace('.txt', '')
    file_path = os.path.join(DATA_DIR, filename)
    chunks = split_document(file_path)
    # 批量构建当前文件的所有数据
    all_chunks.extend(chunks)
    all_metadata.extend([
        {"character": character_name, "source": filename, "chunk_id": i}

    all_ids.extend([f"{character_name}_{i}" for i in range(len(chunks))])

加载知识库
def load_knowledge_base():
    """加载已有的向量知识库"""
    client = chromadb.PersistentClient(path=DB_PATH)
    # 配置Embedding（必须与构建时一致）
    qwen_ef = embedding_functions.OpenAIEmbeddingFunction(
        api_key=API_KEY,
        api_base=API_BASE,
        model_name="text-embedding-v2"
    )
    # 获取集合（不是create，而是get）
    collection = client.get_collection(
        name="genshin_knowledge",
        embedding_function=qwen_ef
    )
    return collection
get 而非 create：使用 get_collection() 而不是 create_collection()，加载已有数据
非常快：加载只需0.1秒，因为不需要向量化，只是读取元数据
非常快
Embedding一致性：查询时使用的Embedding函数必须和构建时一致。因为查询时Chroma会用这
个函数把问题向量化，然后和知识库中的向量计算相似度

def ask_question(collection, query):
    """基于RAG回答问题"""
    # 检索相关文档
    results = collection.query(
        query_texts=[query],
        n_results=3
    )
    retrieved_docs = results['documents'][0]
    # 构造上下文
    context = "\n\n".join([
        f"【参考资料{i+1}】\n{doc}"
        for i, doc in enumerate(retrieved_docs)
    ])
自动向量化：Chroma会自动调用 qwen_ef 把问题向量化
自动向量化
Top-3检索：n_results=3 返回最相关的3个文档块
Top-3检索

def main():
    # 加载知识库（只加载一次）
    collection = load_knowledge_base()
    # 交互式问答
    while True:
        query = input("你：").strip()
        if query in ['退出', 'quit', 'exit']:
            break
        if not query:
            continue
        answer = ask_question(collection, query)
        print(f"\n助手：{answer}\n")
